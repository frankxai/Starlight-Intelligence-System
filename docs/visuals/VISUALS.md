```html
// Intersection + scroll reveal + basic parallax (extend existing initScrollVisuals)
const visuals = document.querySelectorAll('.scroll-visual, .visual-container img');
const observer = new IntersectionObserver((entries) => {
  entries.forEach(e => {
    if (e.isIntersecting) {
      e.target.style.transition = 'transform 0.7s cubic-bezier(0.23,1,0.32,1), opacity 0.7s';
      e.target.style.opacity = '1';

visuals.forEach(v => { v.style.opacity='0.88'; v.style.transform='translateY(28px)'; observer.observe(v); });

// Parallax driver (call on scroll, throttle with rAF)
function applyParallax(container) {
  const layers = container.querySelectorAll('[data-parallax]');
  const rect = container.getBoundingClientRect();
  const progress = Math.max(0, Math.min(1, (window.innerHeight - rect.top) / (window.innerHeight + rect.height)));
  layers.forEach(l => {
    const rate = parseFloat(l.dataset.parallax);
    l.style.transform = `translateY(${ (progress - 0.5) * 60 * rate }px)`;
  });
}
window.addEventListener('scroll', () => document.querySelectorAll('.visual-container').forEach(applyParallax), {passive:true});
</script>
```

**Next.js / React component skeleton (site/src/components/QueenScrollVisual.tsx):**
```tsx
'use client';
import { useEffect, useRef } from 'react';
import Image from 'next/image';

export function QueenScrollVisual({ src, alt, aspect='tall', className='' }: {src:string; alt:string; aspect?:'tall'|'wide'; className?:string}) {
  const ref = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const el = ref.current; if (!el) return;
    const img = el.querySelector('img')!;
    const io = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting) {
        img.style.opacity = '1'; img.style.transform = 'translateY(0)';
      }
    }, {threshold:0.2});
    img.style.opacity = '0.9'; img.style.transform = 'translateY(24px)'; img.style.transition = 'all 700ms cubic-bezier(0.23,1,0.32,1)';
    io.observe(img);
    // add scroll parallax if container has data-layers or children with data-parallax
    return () => io.disconnect();
  }, []);
  return (
    <div ref={ref} className={`visual-container relative overflow-hidden rounded-3xl border border-white/10 ${aspect==='tall' ? 'aspect-[9/16] md:aspect-[3/4]' : 'aspect-video'} ${className}`}>
      <Image src={src} alt={alt} fill className="object-cover" sizes="(max-width:768px) 100vw, 70vw" />
      {/* SIP overlay or caption slot */}
    </div>
  );
}
```
